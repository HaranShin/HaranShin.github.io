# pymssql
 
```python	 
import pandas as pd
from sqlalchemy import create_engine

engine = create_engine(
    f'mssql+pyodbc://사용자명:비번@IP주소,PORT번호/DATABASE명?driver=ODBC+Driver+17+for+SQL+Server', 
    fast_executemany=True   # 대용량 데이터에 적합
    )

df = pd.read_excel(testfile_path, dtype=str)

df.to_sql(
    '테이블명', 
    engine, 
    if_exists='append', 
    index=False,
    chunksize=10000     # 대용량 데이터에 적합
)
```
